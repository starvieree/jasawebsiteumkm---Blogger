<script src="https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore-compat.js"></script>
<script>
(function() {
  var navLayanan = document.getElementById("bw-navbar-layanan");
  if (!navLayanan) return;

  var fbConfig = {
    apiKey: "AIzaSyBDZhr0aTwV67K9RybIW_3HPu1gjebLzAE",
    authDomain: "jasawebsitebatu.firebaseapp.com",
    projectId: "jasawebsitebatu",
    storageBucket: "jasawebsitebatu.firebasestorage.app",
    messagingSenderId: "971578693790",
    appId: "1:971578693790:web:bd2a38586018270201c38f"
  };

  function renderNavbarDropdown(items) {
    if (!items || !items.length) return;
    var html = "";
    items.forEach(function(item) {
      var slug = item.slugId || "layanan";
      var icon = item.icon || "bi-window-sidebar";
      var nama = item.nama || "Layanan";
      html += '<a href="/p/layanan.html#' + encodeURIComponent(slug) + '"><i class="bi ' + escHtml(icon) + '"></i> ' + escHtml(nama) + '</a>';
    });
    navLayanan.innerHTML = html;
  }

  function escHtml(s) {
    return String(s || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
  }

  function syncNavbar() {
    try {
      if (typeof firebase !== 'undefined') {
        if (!firebase.apps.length) {
          firebase.initializeApp(fbConfig);
        }
        var db = firebase.firestore();
        db.collection("layanan").orderBy("urutan", "asc").onSnapshot(function(snap) {
          if (snap.empty) return;
          var list = [];
          snap.forEach(function(doc) {
            list.push(doc.data());
          });
          renderNavbarDropdown(list);
        }, function(err) {
          console.warn("Navbar Firestore listener:", err);
          fetchRestFallback();
        });
        return;
      }
    } catch(e) {}
    fetchRestFallback();
  }

  function fetchRestFallback() {
    fetch("https://firestore.googleapis.com/v1/projects/" + fbConfig.projectId + "/databases/(default)/documents/layanan")
      .then(function(res) { return res.json(); })
      .then(function(data) {
        if (!data || !data.documents) return;
        var list = data.documents.map(function(doc) {
          var f = doc.fields || {};
          return {
            nama: f.nama ? f.nama.stringValue : "",
            slugId: f.slugId ? f.slugId.stringValue : "",
            icon: f.icon ? f.icon.stringValue : "bi-window-sidebar",
            urutan: f.urutan ? (f.urutan.integerValue || f.urutan.doubleValue || 0) : 0
          };
        }).filter(function(i) { return i.nama; });
        list.sort(function(a, b) { return a.urutan - b.urutan; });
        renderNavbarDropdown(list);
      }).catch(function() {});
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", syncNavbar);
  } else {
    syncNavbar();
  }
})();
</script>
