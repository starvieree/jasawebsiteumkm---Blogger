<script>
(function() {
  function updateHomePrices() {
    try {
      var s = localStorage.getItem('bw_firebase_config');
      var cfg = s ? JSON.parse(s) : null;
      if (!cfg || !cfg.apiKey || cfg.apiKey === "ISI_DENGAN_API_KEY_ANDA") return;
      if (typeof firebase === 'undefined') return;
      if (!firebase.apps.length) firebase.initializeApp(cfg);
      var db = firebase.firestore();
      var ul = document.getElementById("bw-home-paket-list");
      if (!db || !ul) return;
      db.collection("paket_harga").orderBy("urutan", "asc").onSnapshot(function(snap) {
        if (snap.empty) return;
        var html = "";
        snap.forEach(function(doc) {
          var d = doc.data();
          var desc = d.deskripsi ? String(d.deskripsi).substring(0, 40) + '...' : '';
          html += '<li style="display:flex;justify-content:space-between;gap:14px"><span><b>' + String(d.nama||'') + '</b><br /><small style="color:var(--mu)">' + desc + '</small></span><b style="color:var(--n);white-space:nowrap">' + String(d.harga||'') + '</b></li>';
        });
        if (html) ul.innerHTML = html;
      });
    } catch(e) {}
  }
  if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', updateHomePrices); } else { updateHomePrices(); }
})();
</script>
